# subdomain_domain_wildcards_image — run commands for every bundled tool

# Image:  subdomain_domain_wildcards_image
# Build:  docker build -t subdomain_domain_wildcards_image \
#           service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/

# -----------------------------------------------------------------------------
# RUN THE PIPELINE, NOT THE TOOLS (read this first)
# -----------------------------------------------------------------------------
# This file exists to reproduce or debug ONE tool by hand. The supported way to
# run this stage is the Python modules, which already do everything below in the
# right order with validated resolvers, wildcard filtering, provenance and a
# machine-readable report:
#
#   # all three stages + the union of live hosts
#   python -m service.recon_pipeline.asset_pipelines.subdomain_domain_wildcards.main -t example.com
#
#   # one stage at a time
#   python -m ...subdomain_domain_wildcards.passive.pipeline     -t example.com
#   python -m ...subdomain_domain_wildcards.active.pipeline      -t example.com
#   python -m ...subdomain_domain_wildcards.permutation.pipeline -t example.com
#
# See README.md, active/README.md and permutation/README.md for flags, outputs
# and measured yields.
#
# -----------------------------------------------------------------------------
# RESOLVERS: THE BUNDLED massdns LIST IS DEAD — DO NOT USE IT
# -----------------------------------------------------------------------------
# The image bundles /usr/local/share/massdns/resolvers.txt (8,477 addresses,
# ~2019 vintage). Measured on 2026-09-16 it resolved NONE of three test names,
# and a hand-checked sample of 60 was ~88% dead. Handing it to puredns/massdns
# silently produces "nothing resolves".
#
# Worse, a single hijacking resolver (captive portal, ISP search page) answers
# names that do not exist — and the tools' wildcard heuristics work by probing a
# random label and expecting NXDOMAIN. Point puredns at a resolver that answers
# everything and it concludes the whole zone is a wildcard and DISCARDS REAL
# RESULTS.
#
# So: let the active stage validate and write the pool, then point the tools at
# that. It probes both halves of the contract (resolves a name that exists; must
# return NXDOMAIN for <random>.invalid) and records rejects with reasons.
#
#   python -m ...subdomain_domain_wildcards.active.pipeline -t example.com --no-bruteforce
#   # -> active/output/resolvers.txt          (the validated pool)
#   # -> active/output/resolvers-trusted.txt  (high-trust subset, for poisoning checks)
#   # -> active/output/resolvers_rejected.txt (with reasons)
#
# Windows / Git Bash notes:
#   - If `-v "$PWD:/work"` fails to mount (path conversion), use
#     `-v "$(pwd -W):/work"` instead.
#   - For absolute paths inside the container, export MSYS_NO_PATHCONV=1 first so
#     Git Bash stops mangling `/work` into `C:/Program Files/Git/work`. The
#     pipeline sets this for its own calls.
#   - Long-running tools can be capped with `timeout 600 docker run ...`
#     (amass has its own `-timeout N`, in minutes).
#   - The image has no ENTRYPOINT — the tool name is the first argument.
# =============================================================================

# -----------------------------------------------------------------------------
# PASSIVE — subdomain enumeration from OSINT / certificate transparency
# -----------------------------------------------------------------------------

# amass (passive) — CT logs + OSINT sources (passive is the default mode).
#
# The image bundles amass v4.2.0, which STREAMS the asset graph to stdout as
# it is built — one relation per line, e.g.:
#   qbsco.net (FQDN) --> a_record --> 104.21.81.2 (IPAddress)
#   qbsco.net (FQDN) --> ns_record --> jason.ns.cloudflare.com (FQDN)
#   13335 (ASN) --> announces --> 172.64.0.0/18 (Netblock)
#
# This "graph relations" output was removed in amass v5 (results go to a
# SQLite asset.db and nothing prints to stdout), which is why this image pins
# v4.2.0. Add `-timeout N` (minutes) to cap the run; first relations usually
# appear after ~2-4 minutes of source setup. `-v` prints which sources are
# being queried (helpful when the output looks thin).
docker run --rm subdomain_domain_wildcards_image amass enum -d example.com -timeout 15

# Save the graph relations to a file on your host:
docker run --rm subdomain_domain_wildcards_image amass enum -d example.com -timeout 15 > amass-graph.txt

# ...or reduce the graph to a plain, deduped subdomain list:
docker run --rm subdomain_domain_wildcards_image amass enum -d example.com -timeout 15 \
  | grep -oE '[a-z0-9.-]+\.example\.com' | sort -u > amass-subdomains.txt

# assetfinder — cert transparency + web sources (--subs-only prints subdomains)
docker run --rm subdomain_domain_wildcards_image assetfinder --subs-only example.com

# chaos — ProjectDiscovery Chaos dataset. Requires a ProjectDiscovery key:
# chaos-client reads PDCP_API_KEY; your key lives in the repo root `.env` as
# CHAOS_API_KEY. Load it from there:
#   KEY=$(grep '^CHAOS_API_KEY=' .env | head -1 | cut -d= -f2- | tr -d '"' | tr -d '\r')
#   docker run --rm -e PDCP_API_KEY=$KEY subdomain_domain_wildcards_image chaos -d example.com -silent
#
# NOTE: not every domain is in the Chaos dataset — qbsco.net returned nothing
# even with a valid key (verified against hackerone.com, which did return data).

# findomain — 30+ passive sources (CT, GitHub, etc.)
docker run --rm subdomain_domain_wildcards_image findomain -t example.com -q

# subfinder — 30+ passive sources, the workhorse
docker run --rm subdomain_domain_wildcards_image subfinder -d example.com -silent

# -----------------------------------------------------------------------------
# MERGE — combine passive results into one deduped working list
# -----------------------------------------------------------------------------
cat findomain.txt assetfinder.txt subfinder.txt 2>/dev/null | sort -u > subdomains.txt

# -----------------------------------------------------------------------------
# ACTIVE — DNS resolution / brute force / zone transfers
# -----------------------------------------------------------------------------

# puredns resolve — the stage's default engine: bulk-resolve a list with
# wildcard filtering + poisoning validation against trusted resolvers.
# `--write-wildcards` records what it decided was a wildcard (auditable).
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  puredns resolve /work/candidates.txt \
    -r /work/resolvers.txt --resolvers-trusted /work/resolvers-trusted.txt \
    --wildcard-tests 3 --write-wildcards /work/puredns-wildcards.txt -w /work/puredns.txt

# puredns bruteforce — wordlist enumeration under one apex (the tool streams the
# wordlist, so a 110k-word SecLists file needs no local pre-processing)
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  puredns bruteforce /work/wordlist.txt example.com \
    -r /work/resolvers.txt --resolvers-trusted /work/resolvers-trusted.txt \
    --write-wildcards /work/puredns-bruteforce-wildcards.txt -w /work/puredns-bruteforce.txt

# shuffledns — the alternative engine. `-mode resolve` takes a name list (`-l`),
# `-mode bruteforce` takes a wordlist (`-w`). `-m` points at the massdns binary.
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  shuffledns -d example.com -r /work/resolvers.txt -m /usr/local/bin/massdns \
    -mode bruteforce -w /work/wordlist.txt -silent -nc -duc

# massdns — raw bulk A resolution (no wildcard handling at all: use it only when
# you have already filtered, or you accept wildcard noise)
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  massdns -r /work/resolvers.txt -t A -o S -w /work/massdns.txt /work/tool-input.txt

# dnsx — record enrichment as JSONL, resolvers EXPLICITLY set:
#   -r  resolver file to use
#   -a -aaaa -cname -ns -mx -txt  the record types this stage asks for
#   -duc disables the self-update check (a run should not need GitHub)
#   WARNING: `-wd` is mutually exclusive with the record-type flags in the
#   bundled version — combining them silently narrows the query set. This stage
#   deliberately does not use it. Also avoid `-recon` here: it fires an AXFR at
#   every host.
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  dnsx -silent -json -l /work/tool-input.txt -r /work/resolvers.txt \
    -a -aaaa -cname -ns -mx -txt -retry 2 -duc -nc > records.jsonl

# dig AXFR — one zone-transfer attempt per nameserver (one query each; almost
# always refused, occasionally an entire authoritative zone). Validate the
# nameserver name before passing it — it comes from a DNS answer.
docker run --rm subdomain_domain_wildcards_image \
  dig +time=10 +tries=1 +noall +answer AXFR example.com @ns1.example.com

# amass (active) — zone transfers + cert name grabs (`-active`; DNS brute forcing
# is a separate flag: `-brute`). Same graph-relations stdout as the passive mode.
docker run --rm subdomain_domain_wildcards_image amass enum -d example.com -active -timeout 15

# httpx — HTTP probing of RESOLVED hosts (status, title, tech, CNAME).
# Opt-in in the pipeline (`--http` / ACTIVE_HTTP=1): this is the only step that
# sends application traffic to the target rather than DNS queries.
docker run --rm -i subdomain_domain_wildcards_image \
  httpx -silent -json -sc -title -td -cname -rl 100 -t 50 -duc -nc \
  < resolved.txt > http.jsonl

# -----------------------------------------------------------------------------
# PERMUTATION — generate new candidate subdomains from known names
# -----------------------------------------------------------------------------

# dnsgen — permutations of known hosts, one per line. DNSGen 2.x takes a FILE
# argument (`-` reads stdin), and PYTHONHASHSEED=0 is what makes the output
# ORDER stable: the tool shuffles its input through sets, so three runs over one
# identical file produce three different orderings — and since the pipeline keeps
# a prefix of that output, an unseeded run makes the cap a lottery. The pipeline
# always sets it; set it by hand too if you care about reproducibility.
PYTHONHASHSEED=0 MSYS_NO_PATHCONV=1 docker run --rm \
  -e PYTHONHASHSEED=0 \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/permutation/output:/work" \
  subdomain_domain_wildcards_image dnsgen /work/known.txt > dnsgen-raw.txt

# ...or feed it from stdin
cat subdomains.txt | docker run --rm -i subdomain_domain_wildcards_image dnsgen -

# Extra words, and a faster/lighter pass:
#   docker run ... dnsgen /work/known.txt --wordlen 4 --wordlist /work/extra-words.txt
#   docker run ... dnsgen /work/known.txt --fast

# -----------------------------------------------------------------------------
# ONE-LINER CHAIN — passive -> permutations -> live host resolution
# -----------------------------------------------------------------------------

# subfinder -> dnsgen -> puredns -> live hosts.
# Note the chain resolves with the VALIDATED pool from the active stage, not the
# bundled list — see the warning at the top of this file.
docker run --rm subdomain_domain_wildcards_image subfinder -d example.com -silent \
  | docker run --rm -i -e PYTHONHASHSEED=0 subdomain_domain_wildcards_image dnsgen - \
  | sort -u > candidates.txt
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  puredns resolve /work/candidates.txt -r /work/resolvers.txt \
    --resolvers-trusted /work/resolvers-trusted.txt -w /work/live-hosts.txt

# -----------------------------------------------------------------------------
# STEALTH: reproducing the shaped run by hand
# -----------------------------------------------------------------------------
# The stage applies all of this itself; these commands exist so a single tool can
# be reproduced with the same shaping. The reasoning, the measurements and the
# limits are in service/recon_pipeline/stealth/README.md.

# 1. The one flag that matters most. httpx enables -random-agent BY DEFAULT,
#    which is what produced user agents like "Firefox/3.6.13" and
#    "Chrome/134.0.0.0" for "Kubuntu; Linux i686" during testing — i.e. a user
#    agent no browser has ever sent, in front of whichever ClientHello the
#    impersonator picked. Off, plus an explicit coherent identity:
docker run --rm subdomain_domain_wildcards_image httpx \
  -u https://example.com -json -irh -random-agent=false \
  -H 'Sec-Ch-Ua: "Chromium";v="134", "Not:A-Brand";v="24", "Google Chrome";v="134"' \
  -H 'Sec-Ch-Ua-Mobile: ?0' -H 'Sec-Ch-Ua-Platform: "Windows"' \
  -H 'Upgrade-Insecure-Requests: 1' \
  -H 'User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/134.0.0.0 Safari/537.36' \
  -H 'Accept: text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,image/apng,*/*;q=0.8' \
  -H 'Sec-Fetch-Site: none' -H 'Sec-Fetch-Mode: navigate' -H 'Sec-Fetch-User: ?1' -H 'Sec-Fetch-Dest: document' \
  -H 'Accept-Encoding: gzip, deflate, br, zstd' -H 'Accept-Language: en-US,en;q=0.9' -H 'Priority: u=0, i'

# 2. TLS impersonation. Verified to produce the real Chrome JA4
#    (t13d1516h2_8daaf6152771_... — the cipher hash Cloudflare publishes for
#    current Chrome). Do NOT combine with -unsafe: raw mode drops the
#    impersonation and the JA4 reverts to Go's default.
docker run --rm subdomain_domain_wildcards_image httpx -u https://example.com -tlsi chrome -silent

# 3. Check any of it yourself (echoes back the JA3/JA4/header order you sent):
#    the Python transport does this too — see stealth/README.md for captures.
docker run --rm subdomain_domain_wildcards_image httpx -u https://tls.peet.ws/api/all -json -irr -tlsi chrome -random-agent=false

# 4. Pace at the volume budget instead of at full speed. massdns/puredns fan out
#    over the resolver file, so per-resolver load is labels/resolvers; the
#    --rate-limit below matches STEALTH_DNS_QPS and the batching the stage does.
MSYS_NO_PATHCONV=1 docker run --rm \
  -v "$(pwd -W)/service/recon_pipeline/asset_pipelines/subdomain_domain_wildcards/active/output:/work" \
  subdomain_domain_wildcards_image \
  puredns bruteforce example.com /work/wordlist.txt -r /work/resolvers.txt \
    --resolvers-trusted /work/resolvers-trusted.txt \
    --rate-limit 100 --wildcard-tests 3 \
    -w /work/bruteforce.txt --write-wildcards /work/bruteforce-wildcards.txt

# 5. Environment knobs (all optional; defaults shown):
#   STEALTH_PASSIVE_ONLY=0               refuse all active technique
#   STEALTH_HOST_QPS=2                   sustained requests/second per host
#   STEALTH_HOST_BURST=4                 per-host burst allowance
#   STEALTH_JITTER=0.35                  timing jitter as a fraction of each delay
#   STEALTH_DNS_NAMES_PER_RESOLVER_HOUR=60   the volume budget per resolver
#   STEALTH_DNS_BATCH_SIZE=500           names per resolver window
#   STEALTH_DNS_BATCH_SPACING=20         jittered seconds between batches
#   STEALTH_DNS_QPS=100                  total query rate across the pool
#   STEALTH_IDENTITY=chrome-win          pin one identity for the whole run
#   STEALTH_QUARANTINE_FILE=<path>       where quarantine state persists
#   ACTIVE_STEALTH=0                     disable the whole layer
#   ACTIVE_HTTP_IMPERSONATE=0            stop passing -tlsi to httpx
#   ACTIVE_AXFR_SPACING=5                seconds between zone-transfer attempts
